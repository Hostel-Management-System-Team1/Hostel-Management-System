34def report_access_required(view_func): 

35 """Restrict reports to wardens and administrators.""" 

36 

37 @login_required 

38 @wraps(view_func) 

39 def wrapped_view(request, *args, **kwargs): 

40 if request.user.role not in ( 

41 User.Role.WARDEN, 

42 User.Role.ADMINISTRATOR, 

43 ): 

44 raise PermissionDenied( 

45 "Only wardens and administrators can access reports." 

46 ) 

47 return view_func(request, *args, **kwargs) 

48 

49 return wrapped_view

149def xlsx_response(filename, rows, sheet_name): 

150 workbook = Workbook() 

151 sheet = workbook.active 

152 sheet.title = sheet_name 

153 

154 for row in rows: 

155 sheet.append(row) 

156 

157 sheet.freeze_panes = "A2" 

158 sheet.auto_filter.ref = sheet.dimensions 

159 

160 for cell in sheet[1]: 

161 cell.font = cell.font.copy(bold=True) 

162 

163 for column_cells in sheet.columns: 

164 width = max( 

165 len(str(cell.value or "")) for cell in column_cells 

166 ) 

167 sheet.column_dimensions[ 

168 column_cells[0].column_letter 

169 ].width = min(width + 2, 40) 

170 

171 output = BytesIO() 

172 workbook.save(output) 

173 output.seek(0) 

174 

175 response = HttpResponse( 

176 output.getvalue(), 

177 content_type=( 

178 "application/vnd.openxmlformats-officedocument." 

179 "spreadsheetml.sheet" 

180 ), 

181 ) 

182 response["Content-Disposition"] = ( 

183 f'attachment; filename="{filename}"' 

184 )

194def _is_number(value): 

195 if isinstance(value, (int, float, Decimal)): 

196 return True 

197 try: 

198 Decimal(str(value)) 

199 except ArithmeticError: 

200 return False 

201 return True

204def pdf_response(filename, title, rows): 

205 """An A4 landscape report that opens in the browser's PDF viewer.""" 

206 output = BytesIO() 

207 page = landscape(A4) 

208 margin = 15 * mm 

209 document = SimpleDocTemplate( 

210 output, pagesize=page, title=title, author="Hostel Management System", 

211 leftMargin=margin, rightMargin=margin, topMargin=margin, bottomMargin=18 * mm, 

212 ) 

213 

214 styles = getSampleStyleSheet() 

215 heading = ParagraphStyle("hms-title", parent=styles["Title"], alignment=0, 

216 fontSize=16, leading=20, textColor=NAVY, spaceAfter=2) 

217 meta = ParagraphStyle("hms-meta", parent=styles["Normal"], fontSize=9, textColor=MUTED) 

218 cell = ParagraphStyle("hms-cell", parent=styles["Normal"], fontSize=8.5, leading=11) 

219 head = ParagraphStyle("hms-head", parent=cell, fontName="Helvetica-Bold", 

220 textColor=colors.white) 

221 

222 header, body = rows[0], rows[1:] 

223 numeric = [ 

224 bool(body) and all(_is_number(row[i]) for row in body if row[i] not in ("", None)) 

225 for i in range(len(header)) 

226 ] 

227 data = [[Paragraph(str(h), head) for h in header]] 

228 for row in body: 

229 data.append([ 

230 value if numeric[i] else Paragraph(escape(str(value)), cell) 

231 for i, value in enumerate(row) 

232 ]) 

233 

234 # Share the width by the longest text in each column (numbers stay narrow). 

235 width = page[0] - 2 * margin 

236 lengths = [ 

237 max([len(str(header[i]))] + [len(str(row[i])) for row in body[:200]]) 

238 for i in range(len(header)) 

239 ] 

240 lengths = [min(max(n, 6), 40) for n in lengths] 

241 widths = [width * n / sum(lengths) for n in lengths] 

242 

243 table = Table(data, colWidths=widths, repeatRows=1) 

244 style = [ 

245 ("BACKGROUND", (0, 0), (-1, 0), NAVY), 

246 ("FONTSIZE", (0, 0), (-1, -1), 8.5), 

247 ("VALIGN", (0, 0), (-1, -1), "MIDDLE"), 

248 ("TOPPADDING", (0, 0), (-1, -1), 5), 

249 ("BOTTOMPADDING", (0, 0), (-1, -1), 5), 

250 ("LINEBELOW", (0, 0), (-1, -1), 0.5, RULE), 

251 ("ROWBACKGROUNDS", (0, 1), (-1, -1), [colors.white, STRIPE]), 

252 ] 

253 style += [("ALIGN", (i, 1), (i, -1), "RIGHT") for i, is_num in enumerate(numeric) if is_num] 

254 table.setStyle(TableStyle(style)) 

255 

256 generated = timezone.localtime().strftime("%d %b %Y, %H:%M") 

257 elements = [ 

258 Paragraph(escape(title), heading), 

259 Paragraph(f"Hostel Management System · generated {generated} · " 

260 f"{len(body)} row{'s' if len(body) != 1 else ''}", meta), 

261 Spacer(1, 6 * mm), 

262 table if body else Paragraph("No records.", meta), 

263 ] 

264 

265 def footer(canvas, doc): 

266 canvas.saveState() 

267 canvas.setFont("Helvetica", 8) 

268 canvas.setFillColor(MUTED) 

269 canvas.drawString(margin, 10 * mm, title) 

270 canvas.drawRightString(page[0] - margin, 10 * mm, f"Page {doc.page}") 

271 canvas.restoreState() 

272 

273 document.build(elements, onFirstPage=footer, onLaterPages=footer) 

274 

275 response = HttpResponse(output.getvalue(), content_type="application/pdf") 

276 response["Content-Disposition"] = f'inline; filename="{filename}"' 

277 return response

280def report_context(): 

281 """Figures and rows shown on the reports page, in the portal and in the admin site.""" 

282 total_beds = Bed.objects.count() 

283 occupied_beds = Bed.objects.filter( 

284 status=Bed.Status.OCCUPIED 

285 ).count() 

286 

287 unpaid_invoices = Invoice.objects.exclude( 

288 status=Invoice.Status.PAID 

289 ) 

290 outstanding_total = ( 

291 unpaid_invoices.aggregate(total=Sum("total"))["total"] or 0 

292 ) 

293 

294 complaints = Complaint.objects.select_related( 

295 "created_by" 

296 ).order_by("status", "created_at") 

297 mess_activities = MessActivity.objects.select_related( 

298 "recorded_by" 

299 ).order_by("-activity_date", "meal_type") 

300 

301 return { 

302 "total_beds": total_beds, 

303 "occupied_beds": occupied_beds, 

304 "vacant_beds": Bed.objects.filter( 

305 status=Bed.Status.VACANT 

306 ).count(), 

307 "occupancy_percent": round( 

308 occupied_beds / total_beds * 100, 1 

309 ) if total_beds else 0, 

310 "unpaid_invoices": unpaid_invoices.select_related("resident").order_by("due_date"), 

311 "outstanding_total": outstanding_total, 

312 "overdue_count": unpaid_invoices.filter(status=Invoice.Status.OVERDUE).count(), 

313 "complaints": complaints, 

314 "open_complaints": complaints.filter( 

315 status__in=[ 

316 Complaint.Status.OPEN, 

317 Complaint.Status.IN_PROGRESS, 

318 ] 

319 ).count(), 

320 "mess_activities": mess_activities[:30], 

321 "total_meals_served": ( 

322 mess_activities.aggregate(total=Sum("meals_served"))["total"] 

323 or 0 

324 ), 

325 "generated_at": timezone.now(), 

326 }
